# train.json and valid.json are lists of objects:
#   {"src_lang": "asm_Beng", "tgt_lang": "eng_Latn", "src_text": "...", "tgt_text": "..."}
# train.py converts them. No separate prepare step was required.

# TRAIN=/shared/nithish/INDICTRANS2_train_custom/data/train.json
# VAL=/shared/nithish/INDICTRANS2_train_custom/data/valid.json


# =============================================================================
# 3. Train — LoRA
#    Saves an adapter at it2_out/lora/final_model
# =============================================================================

python train.py \
  --gpu_ids 0 \
  --model_hf_id ai4bharat/indictrans2-indic-en-dist-200M \
  --train_mode lora \
  --train_json /shared/nithish/INDICTRANS2_train_custom/data/train.json \
  --val_json /shared/nithish/INDICTRANS2_train_custom/data/valid.json \
  --src_lang asm_Beng --tgt_lang eng_Latn \
  --output_dir it2_out/lora \
  --final_model_dir it2_out/lora/final_model \
  --epochs 2 --learning_rate 2e-4 --bf16 \
  --eval_strategy epoch --save_strategy epoch \
  --metric_for_best_model eval_loss --top_n_checkpoints 3


# =============================================================================
# 4. Train — QLoRA
#    4-bit base + LoRA. Saves an adapter at it2_out/qlora/final_model
# =============================================================================

python train.py \
  --gpu_ids 0 \
  --model_hf_id ai4bharat/indictrans2-indic-en-dist-200M \
  --train_mode qlora \
  --train_json /shared/nithish/INDICTRANS2_train_custom/data/train.json \
  --val_json /shared/nithish/INDICTRANS2_train_custom/data/valid.json \
  --src_lang asm_Beng --tgt_lang eng_Latn \
  --output_dir it2_out/qlora \
  --final_model_dir it2_out/qlora/final_model \
  --epochs 2 --learning_rate 2e-4 --bf16 \
  --optim paged_adamw_32bit \
  --bnb_compute_dtype bfloat16 \
  --bnb_4bit_quant_type nf4 \
  --bnb_4bit_use_double_quant \
  --eval_strategy epoch --save_strategy epoch \
  --metric_for_best_model eval_loss --top_n_checkpoints 3


# =============================================================================
# 5. Train — continue the LoRA adapter
#    Starts from it2_out/lora/final_model. Saves it2_out/lora_continue/final_model
# =============================================================================

python train.py \
  --gpu_ids 0 \
  --model_hf_id ai4bharat/indictrans2-indic-en-dist-200M \
  --train_mode lora \
  --adapter_path it2_out/lora/final_model \
  --train_json /shared/nithish/INDICTRANS2_train_custom/data/train.json \
  --val_json /shared/nithish/INDICTRANS2_train_custom/data/valid.json \
  --src_lang asm_Beng --tgt_lang eng_Latn \
  --output_dir it2_out/lora_continue \
  --final_model_dir it2_out/lora_continue/final_model \
  --epochs 2 --learning_rate 2e-4 --bf16 \
  --eval_strategy epoch --save_strategy epoch \
  --metric_for_best_model eval_loss --top_n_checkpoints 3


# =============================================================================
# 6. Train — full_lora
#    LoRA on attention and FFN, then merged. Saves it2_out/full_lora/final_model
# =============================================================================

python train.py \
  --gpu_ids 0 \
  --model_hf_id ai4bharat/indictrans2-indic-en-dist-200M \
  --train_mode full_lora \
  --train_json /shared/nithish/INDICTRANS2_train_custom/data/train.json \
  --val_json /shared/nithish/INDICTRANS2_train_custom/data/valid.json \
  --src_lang asm_Beng --tgt_lang eng_Latn \
  --output_dir it2_out/full_lora \
  --final_model_dir it2_out/full_lora/final_model \
  --epochs 2 \
  --batch_size 1 \
  --gradient_accumulation_steps 8 \
  --learning_rate 2e-5 \
  --optim adamw_bnb_8bit \
  --bf16 \
  --gradient_checkpointing \
  --eval_strategy epoch --save_strategy epoch \
  --metric_for_best_model eval_loss --top_n_checkpoints 3 \
  --save_only_model


# =============================================================================
# 7. Train — full
#    Every parameter. Saves it2_out/full/final_model
# =============================================================================

python train.py \
  --gpu_ids 0 \
  --model_hf_id ai4bharat/indictrans2-indic-en-dist-200M \
  --train_mode full \
  --train_json /shared/nithish/INDICTRANS2_train_custom/data/train.json \
  --val_json /shared/nithish/INDICTRANS2_train_custom/data/valid.json \
  --src_lang asm_Beng --tgt_lang eng_Latn \
  --output_dir it2_out/full \
  --final_model_dir it2_out/full/final_model \
  --epochs 2 \
  --batch_size 1 \
  --gradient_accumulation_steps 8 \
  --learning_rate 2e-5 \
  --optim adamw_bnb_8bit \
  --bf16 \
  --gradient_checkpointing \
  --eval_strategy epoch --save_strategy epoch \
  --metric_for_best_model eval_loss --top_n_checkpoints 3 \
  --save_only_model
