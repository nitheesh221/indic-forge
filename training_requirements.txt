# =============================================================================
# 0b. DATASET FORMAT
# =============================================================================
# File: JSON / JSONL / CSV / TSV
# A JSON document may be:
#   - a list of row objects
#   - JSONL
#   - one object wrapping the list, e.g. {"translation": [ ... ]}
# Required fields per row (aliases allowed):
#   src / src_text / source / source_text     str   source sentence
#   tgt / tgt_text / target / target_text     str   target sentence
# Optional:
#   src_lang      str   FLORES tag (eng_Latn, hin_Deva, asm_Beng, …) or alias (en, hi)
#   tgt_lang      str   FLORES tag
#   id            str
#
# Checked-in sample of that wrapped form:
#   sample_train_data/train.json
#   sample_train_data/valid.json
#   11 rows, asm_Beng -> eng_Latn
#
# Each row keeps its own src_lang and tgt_lang. A file may mix the scheduled
# Indic languages, for example ben_Beng and kan_Knda in the same JSON.
# Omit --src_lang and --tgt_lang to train every pair whose direction matches
# the checkpoint. An indic-en model keeps rows whose target is eng_Latn and
# skips a row aimed the other way. Passing both flags still trains one pair.
# A row with no language tags uses the flags; if those are also omitted, the
# row is skipped.
# Internally converted to official IT2 pair folders:
#   {data_dir}/train/{src}-{tgt}/train.{src}
#   {data_dir}/train/{src}-{tgt}/train.{tgt}
#   {data_dir}/dev/{src}-{tgt}/dev.{src}
#   {data_dir}/dev/{src}-{tgt}/dev.{tgt}
#
# Loss: standard seq2seq CE on target tokens (labels, pad = -100).
# Preprocess: IndicProcessor (inference=False) then HF tokenizer, max_length=256.

# =============================================================================
# 0c. TRAIN MODES  (same semantics as Qwen)
# =============================================================================
# train_mode | what trains                         | --final_model_dir
# -----------|-------------------------------------|---------------------------
# full       | all base weights                    | full HF model
# full + --adapter_path | merge adapter, then full | full HF model
# full_lora  | LoRA on ALL linear attn+FFN modules | merged full model
# full_lora + --adapter_path | continue/merge LoRA | merged full model
# lora       | LoRA on official IT2 modules only   | adapter files
# lora + --adapter_path | continue existing LoRA   | adapter files
# qlora      | 4-bit frozen base + LoRA            | adapter files
# qlora + --adapter_path | continue existing QLoRA | adapter files
#
# Not supported: full_qlora
#
# LoRA target_modules (MODEL-SPECIFIC, IndicTrans2 M2M-style):
#   lora:       q_proj, k_proj          # official AI4Bharat HF recipe
#   full_lora:  q_proj, k_proj, v_proj, out_proj, fc1, fc2
#   qlora:      same module set as lora (override with --lora_target_modules)

# =============================================================================
# 0d. LAUNCH  (same rules as Qwen)
# =============================================================================

#          (omit --gpu_ids when using torchrun)
# nproc_per_node must equal the number of IDs in CUDA_VISIBLE_DEVICES.
# Do not mix python --gpu_ids A,B auto-relaunch with torchrun + --gpu_ids.
# Entry point: train.py  (CLI: indictrans2-ft train)

#   required: no
#   type: str
#   choices: full | full_lora | lora | qlora

# --adapter_path
#   required: no
#   type: str
#   default: None
#   notes: existing LoRA/QLoRA dir; merge/continue depending on train_mode

# --model_hf_id
#   required: yes
#   type: str
#   choices (architecture family + tokenizer):
#     ai4bharat/indictrans2-en-indic-dist-200M
#     ai4bharat/indictrans2-en-indic-1B
#     ai4bharat/indictrans2-indic-en-dist-200M
#     ai4bharat/indictrans2-indic-en-1B
#     ai4bharat/indictrans2-indic-indic-dist-320M
#     ai4bharat/indictrans2-indic-indic-1B
#   default: none
#   notes: MODEL-SPECIFIC. Must match translation direction:
#            English → Indic  => en-indic
#            Indic → English  => indic-en
#            Indic → Indic    => indic-indic
#          trust_remote_code=True

# --model_id
#   required: no
#   type: str
#   default: same as --model_hf_id
#   notes: Hub id or local weights path actually loaded

# --src_lang / --tgt_lang
#   required: no
#   type: str  (FLORES or alias)
#   default: unset
#   notes: optional filters. Unset trains every JSON pair that matches the
#          checkpoint direction (en-indic, indic-en, or indic-indic).
#          One shared target such as eng_Latn can have many source languages
#          in the same run. Each sentence is preprocessed with its own tags.
#          Validation metrics stay one combined score. Set both flags to
#          train only that pair. An explicit pair whose direction does not
#          match the checkpoint is rejected.

# --output_dir
#   required: no
#   default: ./it2_out/exp

# --final_model_dir
#   required: no
#   default: ./it2_out/exp/final_model

#   required: no

# =============================================================================
# 8. TRANSLATION METRICS (CSV + eval command; do NOT select checkpoints)
# =============================================================================
# Same role as Qwen OCR metrics: reported on generated val translations.
# Priority (user-locked):
#   1 chrF++     Best primary metric (official IT2 / Indic morphology)
#   2 COMET      Best semantic-quality metric (src+hyp+ref)
#   3 BLEU       Standard MT benchmark (sacrebleu flores200)
#   4 BERTScore  Additional semantic evaluation (multilingual encoder)
#   5 METEOR     Additional lexical/semantic (weaker on Indic WordNet)
#   6 TER        Edit-distance / error analysis
#   7 Exact match Strict sentence-level diagnostic for short/templated text
#
# --eval_metrics  nargs=+  default: all seven in that order
#   chrf2pp       sacrebleu CHRF(word_order=2) = chrF++
#   comet         Unbabel/wmt22-comet-da (download on first eval)
#   bleu          sacrebleu BLEU tokenize=flores200
#   bertscore     bert-score, model xlm-roberta-large (Indic-capable)
#   meteor        evaluate/nltk METEOR
#   ter           sacrebleu TER
#   exact_match   percentage of hypotheses exactly matching references after
#                 Unicode NFC normalization, whitespace collapse, and strip;
#                 case and punctuation remain significant
#
# Checkpoint selection stays eval_loss (not these).
# If COMET/BERTScore fail (no GPU / download), still write sacrebleu metrics.
#
# --eval_metrics_csv    default: eval_metrics.csv   under --output_dir
# --eval_samples_csv    default: eval_samples.csv   src, ref, hyp per row
# --training_logs_csv   default: training_logs.csv

# 18. INFERENCE
# =============================================================================
# Entry:
#   PYTHONPATH=src python -m indictrans2_finetune.cli translate ...
#   indictrans2-ft translate ...
# Batch script for every saved mode: scripts/inference.sh
# Command log: commands.txt
#
# Three input forms. Pass only one:
#   --text        one sentence. Prints the translation. No metrics.
#   --file        one source sentence per line. Prints translations. No metrics.
#   --json-file   same JSON as train/val. src_text is the input.
#                 tgt_text is the reference and is scored.
#
# --src-lang / --tgt-lang
#   required with --text and --file
#   optional with --json-file when every row already has the tags
#   when both are passed with JSON, only that pair is scored; other pairs are skipped
#
# --model
#   Hub id or a local merged directory.
#   lora / qlora / continued lora:
#     --model ai4bharat/indictrans2-indic-en-dist-200M
#     --adapter it2_out/<mode>/final_model
#   full_lora / full:
#     --model it2_out/<mode>/final_model
#     do not pass --adapter
#   A merged directory with finetune_meta.json loads the tokenizer and the
#   initialized base from model_hf_id, then overlays the saved trained weights.
#   QLoRA is not reloaded in 4-bit at inference. The adapter is applied to the
#   normal base checkpoint.
#
# --adapter
#   default: none
#   directory containing adapter_config.json and adapter weights
#
# --eval-metrics
#   default when omitted on --json-file: all seven, same order as training
#     chrf2pp comet bleu bertscore meteor ter exact_match
#   pass one or more names to score only those
#   spaces or commas both work
#   an unknown name stops before generation
#   one metric that fails (download, missing package) is stored in metric_errors
#   and the other selected metrics are still written
#   these scores do not train and do not select checkpoints
#
# --predictions-csv
#   default: inference_predictions.csv
#   one row per sentence: src_lang, tgt_lang, src, reference, prediction
#
# --metrics-csv
#   default: inference_metrics.csv
#   one corpus row: src_lang, tgt_lang, n, model, adapter, selected scores, metric_errors
#
# --max-length   default: 256
# --num-beams    default: 5
# --batch-size   default: 8
#   sentences per model.generate call for translate and eval
#   order matches the input, so prediction rows stay aligned with references
#   training validation still uses --eval_batch_size (default 2), which is separate
#
# JSON and text-file inference are generated in batches of --batch-size.
# A file shorter than the batch size is one generate call.
#
# COMET scoring loads Unbabel/wmt22-comet-da and runs on CPU (gpus=0).
# "Encoder model frozen" and "GPU available: True, used: False" come from that
# COMET load. They do not freeze IndicTrans2 and they do not move translation
# off the GPU. NLTK may warn about a group-writable nltk_data directory; METEOR
# still scores.
#
# Example, LoRA:
# PYTHONPATH=src python -m indictrans2_finetune.cli translate \
#   --json-file sample_train_data/valid.json \
#   --src-lang asm_Beng --tgt-lang eng_Latn \
#   --model ai4bharat/indictrans2-indic-en-dist-200M \
#   --adapter it2_out/lora/final_model \
#   --eval-metrics chrf2pp comet bleu bertscore meteor ter exact_match \
#   --predictions-csv it2_out/lora/inference_predictions.csv \
#   --metrics-csv it2_out/lora/inference_metrics.csv \
#   --max-length 256 --num-beams 5 --batch-size 8
#
# Same command for qlora and lora_continue, changing --adapter and the CSV paths.
# full_lora and full drop --adapter and set --model to it2_out/<mode>/final_model.
